\vspace{4pt}
\textbf{Processing time rule:}
\begin{equation*}
  p(i,r,j) =
  \begin{cases}
    p_{\text{cycle}}(i,r) / 60 & \text{if } j \text{ is the bottleneck machine of route } r \\
    p_{\text{cycle}}(i,r) / 60 \times 0.25 & \text{otherwise (support operation)}
  \end{cases}
\end{equation*}

\textbf{Setup time rule:}
\begin{equation*}
  s(i,i',j) =
  \begin{cases}
    40 \text{ min} & \text{if jobs } i,i' \text{ differ in ruling type and } j \in \{\text{Bielom1},\text{Bielom2}\} \\
    0              & \text{otherwise}
  \end{cases}
\end{equation*}

% ── Objective Function ──────────────────────────────────────────────────────
\section{Objective Function}

\begin{mdframed}[backgroundcolor=lightblue, linecolor=headerblue, linewidth=1.5pt,
                  innertopmargin=8pt, innerbottommargin=8pt]
\begin{equation}
  \min\; Z = \underbrace{\alpha \cdot C_{\max}}_{\text{makespan}}
            + \underbrace{\beta \cdot \sum_{i \in \mathcal{I}} T_i(i)}_{\text{total tardiness}}
            + \underbrace{\gamma \cdot \sum_{i \in \mathcal{I}} \delta(i)}_{\text{late job count}}
\end{equation}
\textbf{Implemented weights:}\quad
$\alpha = 0.10$,\quad $\beta = 0.60$,\quad $\gamma = 0.30$,\quad
subject to $\alpha + \beta + \gamma = 1$.
\end{mdframed}

\vspace{4pt}
\begin{tabularx}{\linewidth}{>{\bfseries}l l X}
  \toprule
  \rowcolor{lightblue}
  Term & Weight & Purpose \\
  \midrule
  $\alpha \cdot C_{\max}$ & 0.10 & Minimize total production span; earlier batch completion \\
  \rowcolor{rowgray}
  $\beta \cdot \sum T_i$  & 0.60 & Penalise cumulative lateness; critical for priority orders \\
  $\gamma \cdot \sum \delta_i$ & 0.30 & Minimise number of customers affected by late delivery \\
  \bottomrule
\end{tabularx}

% ── Constraints ─────────────────────────────────────────────────────────────
\section{Constraints}

\subsection{K1 — Unique Route Assignment}
\begin{equation}
  \sum_{r \in \mathcal{R}:\, \mathit{compat}(i,r)=1} x(i,r) = 1
  \qquad \forall\, i \in \mathcal{I},\; \mathit{qty}(i)>0
\end{equation}
Every job is assigned to \emph{exactly one} compatible production route.

\subsection{K2 — Route–Machine Consistency}
\begin{equation}
  y(i,j) = \sum_{\substack{r:\, \mathit{ar}(r,j)=1 \\ \mathit{compat}(i,r)=1}} x(i,r)
  \qquad \forall\, i \in \mathcal{I},\; j \in \mathcal{J}
\end{equation}
Machine usage is fully determined by the route assignment; no independent choice.

\subsection{K4 — Completion Time Lower Bound}
\begin{equation}
  C(i,j) \;\geq\; \sum_{\substack{r:\, \mathit{compat}(i,r)=1 \\ \mathit{ar}(r,j)=1}}
    p(i,r,j)\cdot \mathit{qty}(i)\cdot x(i,r)
  \qquad \forall\, i \in \mathcal{I},\; j \in \mathcal{J}
\end{equation}
Completion time on a machine is at least the job's own processing time on that machine.

\subsection{K5 — No-Overlap (Disjunctive) Constraints}
\begin{align}
  C(i,j) &\geq C(i',j)
    + \sum_{r} p(i,r,j)\cdot\mathit{qty}(i)\cdot x(i,r)
    + \sum_{r} s(i',i,j)
    - M\bigl(1 - z(i,i',j)\bigr)
  &&\forall\, i \neq i',\; j \label{eq:K5a}\\[4pt]
  C(i',j) &\geq C(i,j)
    + \sum_{r} p(i',r,j)\cdot\mathit{qty}(i')\cdot x(i',r)
    + \sum_{r} s(i,i',j)
    - M\cdot z(i,i',j)
  &&\forall\, i \neq i',\; j \label{eq:K5b}
\end{align}
Exactly one ordering is active per pair. When $z(i,i',j)=1$: job $i$ completes before $i'$ starts (Eq.~\ref{eq:K5a} active). When $z(i,i',j)=0$: job $i'$ goes first (Eq.~\ref{eq:K5b} active).

\subsection{K6 — Job and Makespan Completion}
\begin{align}
  C_i(i) &\geq C(i,j) &&\forall\, i \in \mathcal{I},\; j : y(i,j)=1 \\
  C_{\max} &\geq C_i(i) &&\forall\, i \in \mathcal{I}
\end{align}
A job is done when it clears its \emph{last} machine. Makespan is the latest over all jobs.

\subsection{K7 — Tardiness Definition}
\begin{align}
  T_i(i) &\geq C_i(i) - d(i) &&\forall\, i \in \mathcal{I} \\
  T_i(i) &\geq 0               &&\forall\, i \in \mathcal{I} \\
  \delta(i) &\geq \frac{C_i(i) - d(i)}{M} &&\forall\, i \in \mathcal{I}
\end{align}
$T_i$ records minutes past deadline; $\delta(i)$ becomes 1 whenever any tardiness exists.

\subsection{K8 — In-Route Machine Precedence}
\begin{equation}
  C(i,j') \geq C(i,j) + p(i,r,j)\cdot\mathit{qty}(i)\cdot x(i,r)
\end{equation}
for all $i,r,j,j'$ where $\mathit{compat}(i,r)=1$, $\mathit{ar}(r,j)=\mathit{ar}(r,j')=1$,
and $\mathit{seq}(r,j')=\mathit{seq}(r,j)+1$.\\
A job cannot start on the next machine in its route until it finishes on the current one.

\subsection{K9 — Variable Domains}
\begin{equation}
  x(i,r),\;\delta(i),\;z(i,i',j) \in \{0,1\};
  \qquad
  y(i,j),\;C(i,j),\;C_i(i),\;T_i(i),\;C_{\max} \geq 0
\end{equation}
